cmake_minimum_required(VERSION 3.22)
# Preserve independent subdirectory builds from the original project.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(dg_hdg LANGUAGES CXX)
    if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
        set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
    endif()
    include(CTest)
    include(../cmake/DghBuild.cmake)
endif()

file(GLOB_RECURSE solver_sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp")
list(REMOVE_ITEM solver_sources "${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp")
add_library(euler_1d ${solver_sources})
dgh_add_common(euler_1d)
target_include_directories(euler_1d PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(euler_1d PUBLIC dgh_build_options dgh_fastgl)
dgh_add_solver(dg_hdg src/main.cpp)
target_link_libraries(dg_hdg PRIVATE euler_1d)

if(BUILD_TESTING)
    add_executable(check_1d check/main.cpp check/check_1d.cpp check/linearization.cpp
                   check/dg_reference.cpp)
    target_include_directories(check_1d PRIVATE check)
    target_link_libraries(check_1d PRIVATE euler_1d)
    add_test(NAME dg_hdg_1d_checks COMMAND check_1d)
    add_executable(benchmark_dg check/benchmark_dg.cpp check/dg_reference.cpp)
    target_link_libraries(benchmark_dg PRIVATE euler_1d)
endif()
