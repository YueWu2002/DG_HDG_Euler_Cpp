cmake_minimum_required(VERSION 3.22)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(dg_hdg_2d LANGUAGES CXX)
    if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
        set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
    endif()
    include(CTest)
    include(../cmake/DghBuild.cmake)
endif()
file(GLOB solver_sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp")
list(REMOVE_ITEM solver_sources "${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp")
add_library(euler_2d ${solver_sources})
dgh_add_common(euler_2d)
target_include_directories(euler_2d PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(euler_2d PUBLIC dgh_build_options dgh_fastgl)
dgh_add_solver(dg_hdg_2d src/main.cpp)
target_link_libraries(dg_hdg_2d PRIVATE euler_2d)
if(BUILD_TESTING)
    add_executable(check_2d check/check_2d.cpp)
    target_link_libraries(check_2d PRIVATE euler_2d)
    add_test(NAME dg_hdg_2d_checks COMMAND check_2d)
endif()
